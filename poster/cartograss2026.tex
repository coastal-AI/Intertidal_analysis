% Cartagena, 24-26 November 2026 (posters only, A0 vertical).
%
% Structure of the reference poster Jorge chose (Oiry et al., World Seagrass
% Conference 2024): rows of panels, short sentences, big images.
%
%   1  header band        conference mark | title, kicker, authors | logos
%   2  context row        text + photo | image pair | text
%   3  concept row        concept | tide figure | OBJECTIVE | pilot result + QR
%   4  materials/methods  sites, data, a pixel is a gauge | how MAREA works
%   5  results            RESERVED grid, filled after the validation
%   6  conclusions        RESERVED banner
%   7  bottom strip       acknowledgments, credits, references, QR
%
% Every row sits on one four-column grid (A 165 | B 211 | C 211 | D 165 mm,
% 15 mm gutters, 22 mm margins), the bottom strip included.
%
% Same family as poster.tex: its preamble files are reused unchanged; the new
% macros (\cgpanel, \cgfig, \cgreserved, ...) live in content/cg/cg_macros.tex.
% The prose is defined in content/cg/text_*.tex and placed by the blocks in
% content/cg/blk_*.tex; figures are read from figures/cg/.
%
% Build: python poster/tools/cg/build_poster.py (from the repository root).
% Two pdflatex passes: every panel is placed with `remember picture'. The log
% lines CGPANEL / CGROW / CGFIG / CGTEXT report the geometry and any overflow.
% usepdftitle=false: beamer would otherwise write its own (empty) \title into
% the PDF metadata over the one set with \hypersetup below.

\newcommand{\cgdef}[2]{\newlength{#1}\setlength{#1}{#2}}

% --- horizontal grid: four columns, 22 mm side margins, 15 mm gutters ----
\cgdef{\cgMargin}{22mm}
\cgdef{\cgGut}{15mm}
\cgdef{\wFull}{797mm}                                        % 841 - 2 x 22
\cgdef{\wA}{165mm}  \cgdef{\xA}{\cgMargin}                   % x = 22
\cgdef{\wB}{211mm}  \cgdef{\xB}{\dimexpr\xA+\wA+\cgGut\relax} % 202
\cgdef{\wC}{211mm}  \cgdef{\xC}{\dimexpr\xB+\wB+\cgGut\relax} % 428
\cgdef{\wD}{165mm}  \cgdef{\xD}{\dimexpr\xC+\wC+\cgGut\relax} % 654
\cgdef{\wHalf}{\dimexpr\wA+\cgGut+\wB\relax}               % 391 = A+B = C+D
\cgdef{\wPair}{\dimexpr\wB+\cgGut+\wC\relax}               % 437 = B+C
\cgdef{\wSub}{\dimexpr(\wHalf-\cgGut)/2\relax}             % 188: half of a half
\cgdef{\xSubB}{\dimexpr\xA+\wSub+\cgGut\relax}

% --- vertical budget -------------------------------------------------
% 2026-10-19: the method flow chart needed 12 mm more than the 252 mm the
% Methods row had. They come from the row gaps (11 -> 10 mm, six gaps) and
% the conclusions banner (46 -> 40 mm, padding 9 -> 6 mm); every Results
% panel keeps its exact size (\hRes stays 286 mm: see the CGPANEL res-* lines).
\cgdef{\gapRow}{10mm}     % between rows
\cgdef{\gapIn}{12mm}      % between panels stacked inside a row
\cgdef{\hHead}{118mm}     % header band, full bleed
\cgdef{\hCtx}{149mm}      % context row: the image pair and its caption line
\cgdef{\hCon}{146mm}      % concept row: the tide figure and its caption line
\cgdef{\hSec}{20mm}       % a section heading over its rule
\cgdef{\gapSec}{6mm}      % heading rule to the panels under it
\cgdef{\hMM}{264mm}       % materials and methods panels
\cgdef{\hConc}{40mm}      % conclusions banner
\cgdef{\hFoot}{74mm}      % bottom strip, full bleed to the bottom edge

\cgdef{\yCtx}{\dimexpr\hHead+\gapRow\relax}
\cgdef{\yCon}{\dimexpr\yCtx+\hCtx+\gapRow\relax}
\cgdef{\ySecMM}{\dimexpr\yCon+\hCon+\gapRow\relax}
\cgdef{\yMM}{\dimexpr\ySecMM+\hSec+\gapSec\relax}
\cgdef{\ySecRes}{\dimexpr\yMM+\hMM+\gapRow\relax}
\cgdef{\yRes}{\dimexpr\ySecRes+\hSec+\gapSec\relax}
\cgdef{\yFoot}{\dimexpr1189mm-\hFoot\relax}
\cgdef{\yConc}{\dimexpr\yFoot-\gapRow-\hConc\relax}
% The results grid takes whatever the rows above and below leave.
\cgdef{\hRes}{\dimexpr\yConc-\gapRow-\yRes\relax}

% --- materials and methods: left half = the study sites and the data (one
%     panel), then one real pixel; right half = the method flow chart, the
%     whole height of the row
\cgdef{\hMat}{176.5mm}
\cgdef{\gapPix}{10mm}     % the materials panel to the pixel panel under it
\cgdef{\yPix}{\dimexpr\yMM+\hMat+\gapPix\relax}
\cgdef{\hPix}{\dimexpr\hMM-\hMat-\gapPix\relax}

% --- results: left half = the Ems maps (half width, room for three maps side
%     by side) over two small panels; right half = the emerged-time map (the
%     largest slot) over a wide, low panel. Each slot is its panel's height
%     less 36 mm: the title and a one-line takeaway to come.
%     2026-10-19 (round 4): the two small panels sit on the grid, A | B
%     (165 | 211 mm, gutter at 187-202 mm like every other row that splits A
%     from B), not as equal halves of the half (188 | 188, gutter at
%     210-225 mm): the result figures are exported at these slot sizes.
\cgdef{\hResC}{148mm}
\cgdef{\yResAB}{\dimexpr\yRes+\hResC+\cgGut\relax}
\cgdef{\hResAB}{\dimexpr\hRes-\hResC-\cgGut\relax}
\cgdef{\hResE}{84mm}
\cgdef{\hResD}{\dimexpr\hRes-\hResE-\cgGut\relax}
\cgdef{\yResE}{\dimexpr\yRes+\hResD+\cgGut\relax}
\cgdef{\figResCH}{\dimexpr\hResC-36mm\relax}
\cgdef{\figResABH}{\dimexpr\hResAB-36mm\relax}
\cgdef{\figResDH}{\dimexpr\hResD-36mm\relax}
\cgdef{\figResEH}{\dimexpr\hResE-36mm\relax}

% --- bottom strip, inside the band (12 mm kept clear of the trim edge), on
%     the grid: acknowledgments A, data B, references C (to the QR), QR at
%     the right edge of D
\cgdef{\yFootIn}{\dimexpr\yFoot+10mm\relax}
\cgdef{\hFootIn}{\dimexpr\hFoot-10mm-12mm\relax}
\cgdef{\wAck}{\wA}
\cgdef{\wCred}{\wB}    \cgdef{\xCred}{\xB}
